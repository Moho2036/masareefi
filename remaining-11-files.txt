import React, { useState, useRef, useMemo } from 'react'
import { format } from 'date-fns'
import { Check, Trash2, AlertCircle, ArrowLeft } from 'lucide-react'
import { Link } from 'react-router'
import { SegmentedControl } from './ui/SegmentedControl'
import { Button } from './ui/Button'
import { Input } from './ui/Input'
import { Toggle } from './ui/Toggle'
import { useAuth } from '../hooks/useAuth'
import { useCategories } from '../hooks/useCategories'
import { useSelectedMonth } from '../hooks/useSelectedMonth'
import { useToast } from '../hooks/useToast'
import { useCurrency } from '../hooks/useCurrency'
import {
  addTransaction,
  updateTransaction,
  deleteTransaction,
} from '../services/transactions'
import { addRecurring } from '../services/recurring'
import { advanceDate } from '../lib/recurring'
import { toFils, fromFils } from '../lib/money'
import { parseLocalDateInput } from '../lib/dates'
import {
  getCategoryUsageMap,
  recordCategoryUsage,
  sortCategoriesByUsage,
} from '../lib/categoryUsage'
import { cn } from '../lib/cn'
import type { Transaction, TxType, Frequency } from '../types'

export interface TransactionFormProps {
  initialTransaction?: Transaction | null
  duplicateFrom?: Transaction | null
  initialType?: TxType
  onClose: () => void
  onSuccess?: () => void
}

export const TransactionForm: React.FC<TransactionFormProps> = ({
  initialTransaction,
  duplicateFrom,
  initialType = 'expense',
  onClose,
  onSuccess,
}) => {
  const { user } = useAuth()
  const { data: categories, loading: categoriesLoading } = useCategories()
  const { setSelectedDate, range } = useSelectedMonth()
  const { showToast } = useToast()
  const { config: currencyConfig } = useCurrency()

  const isEditing = Boolean(initialTransaction)
  const isLinkedToDebt = Boolean(
    initialTransaction?.id &&
      (initialTransaction.id.startsWith('debt_init_') ||
        initialTransaction.id.startsWith('debt_pay_'))
  )
  const amountInputRef = useRef<HTMLInputElement>(null)

  const [type, setType] = useState<TxType>(() => {
    if (initialTransaction) return initialTransaction.type
    if (duplicateFrom) return duplicateFrom.type
    return initialType
  })

  const [amountStr, setAmountStr] = useState<string>(() => {
    if (initialTransaction) return String(fromFils(initialTransaction.amountFils))
    if (duplicateFrom) return String(fromFils(duplicateFrom.amountFils))
    return ''
  })

  const [categoryId, setCategoryId] = useState<string>(() => {
    if (initialTransaction) return initialTransaction.categoryId
    if (duplicateFrom) return duplicateFrom.categoryId
    return ''
  })

  const [dateStr, setDateStr] = useState<string>(() => {
    if (initialTransaction?.date) {
      const d =
        initialTransaction.date instanceof Date
          ? initialTransaction.date
          : new Date(initialTransaction.date)
      return format(d, 'yyyy-MM-dd')
    }
    const now = new Date()
    const isWithinRange = now >= range.start && now < range.end
    return format(isWithinRange ? now : range.start, 'yyyy-MM-dd')
  })

  const [title, setTitle] = useState<string>(
    initialTransaction?.title ?? duplicateFrom?.title ?? ''
  )
  const [note, setNote] = useState<string>(
    initialTransaction?.note ?? duplicateFrom?.note ?? ''
  )
  const [showNoteField, setShowNoteField] = useState<boolean>(() =>
    Boolean(initialTransaction?.note || duplicateFrom?.note)
  )
  const [isRecurring, setIsRecurring] = useState(false)
  const [recurringFrequency, setRecurringFrequency] = useState<Frequency>('monthly')
  const [error, setError] = useState<string | null>(null)
  const [submitting, setSubmitting] = useState(false)
  const [showDeleteConfirm, setShowDeleteConfirm] = useState(false)
  const [deleting, setDeleting] = useState(false)

  // Track category usage frequencies to prioritize top categories (read once on open)
  const usageMap = useMemo(() => getCategoryUsageMap(), [])

  // Filter & sort available categories for current transaction type
  const availableCategories = useMemo(() => {
    const filtered = categories.filter((c) => c.type === type && !c.archived)
    return sortCategoriesByUsage(filtered, usageMap)
  }, [categories, type, usageMap])

  // Compute the effectively selected category ID:
  // If user selected one that belongs to available categories, use it.
  // Otherwise, fallback to the top (most frequent) available category.
  const activeCategoryId =
    categoryId && availableCategories.some((c) => c.id === categoryId)
      ? categoryId
      : availableCategories[0]?.id ?? ''

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault()
    setError(null)

    if (!user) {
      setError('يجب تسجيل الدخول أولاً')
      return
    }

    const amountFils = toFils(amountStr)
    if (amountFils <= 0) {
      setError('يرجى إدخال مبلغ صحيح أكبر من الصفر')
      return
    }

    if (!activeCategoryId) {
      setError('يرجى اختيار التصنيف')
      return
    }

    const parsedDate = parseLocalDateInput(dateStr)
    if (isNaN(parsedDate.getTime())) {
      setError('يرجى اختيار تاريخ صالح')
      return
    }

    setSubmitting(true)
    try {
      const trimmedTitle = title.trim().slice(0, 100)
      if (isEditing && initialTransaction) {
        const finalType = isLinkedToDebt ? initialTransaction.type : type
        const finalAmountFils = isLinkedToDebt ? initialTransaction.amountFils : amountFils
        await updateTransaction(user.uid, initialTransaction.id, {
          type: finalType,
          amountFils: finalAmountFils,
          categoryId: activeCategoryId,
          date: parsedDate,
          title: trimmedTitle,
          note: note.trim().slice(0, 500),
        })
        recordCategoryUsage(activeCategoryId)
        showToast('تم تعديل المعاملة بنجاح', 'success')
      } else {
        let recurringId: string | undefined
        if (isRecurring) {
          const nextDate = advanceDate(parsedDate, recurringFrequency)
          recurringId = await addRecurring(user.uid, {
            type,
            amountFils,
            categoryId: activeCategoryId,
            ...(trimmedTitle ? { title: trimmedTitle } : {}),
            note: note.trim().slice(0, 500),
            frequency: recurringFrequency,
            startDate: parsedDate,
            nextDate,
            active: true,
          })
        }

        await addTransaction(user.uid, {
          type,
          amountFils,
          categoryId: activeCategoryId,
          date: parsedDate,
          ...(trimmedTitle ? { title: trimmedTitle } : {}),
          note: note.trim().slice(0, 500),
          ...(recurringId ? { recurringId } : {}),
        })

        recordCategoryUsage(activeCategoryId)

        showToast(
          duplicateFrom
            ? 'تم حفظ المعاملة المكررة بنجاح'
            : isRecurring
              ? 'تمت إضافة المعاملة وتفعيل التكرار بنجاح'
              : 'تمت إضافة المعاملة بنجاح',
          'success'
        )
      }

      if (parsedDate < range.start || parsedDate >= range.end) {
        setSelectedDate(parsedDate)
      }

      onSuccess?.()
      onClose()
    } catch (err: unknown) {
      console.error('Failed to save transaction:', err)
      setError('تعذر حفظ المعاملة. يرجى التحقق من البيانات والمحاولة مجدداً.')
    } finally {
      setSubmitting(false)
    }
  }

  const handleSaveAndAddAnother = async () => {
    setError(null)

    if (!user) {
      setError('يجب تسجيل الدخول أولاً')
      return
    }

    const amountFils = toFils(amountStr)
    if (amountFils <= 0) {
      setError('يرجى إدخال مبلغ صحيح أكبر من الصفر')
      return
    }

    if (!activeCategoryId) {
      setError('يرجى اختيار التصنيف')
      return
    }

    const parsedDate = parseLocalDateInput(dateStr)
    if (isNaN(parsedDate.getTime())) {
      setError('يرجى اختيار تاريخ صالح')
      return
    }

    setSubmitting(true)
    try {
      const trimmedTitle = title.trim().slice(0, 100)
      let recurringId: string | undefined
      if (isRecurring) {
        const nextDate = advanceDate(parsedDate, recurringFrequency)
        recurringId = await addRecurring(user.uid, {
          type,
          amountFils,
          categoryId: activeCategoryId,
          ...(trimmedTitle ? { title: trimmedTitle } : {}),
          note: note.trim().slice(0, 500),
          frequency: recurringFrequency,
          startDate: parsedDate,
          nextDate,
          active: true,
        })
      }

      await addTransaction(user.uid, {
        type,
        amountFils,
        categoryId: activeCategoryId,
        date: parsedDate,
        ...(trimmedTitle ? { title: trimmedTitle } : {}),
        note: note.trim().slice(0, 500),
        ...(recurringId ? { recurringId } : {}),
      })

      recordCategoryUsage(activeCategoryId)

      showToast('تم حفظ المعاملة بنجاح، وجاهز للمعاملة التالية', 'success')

      if (parsedDate < range.start || parsedDate >= range.end) {
        setSelectedDate(parsedDate)
      }

      onSuccess?.()

      // Reset fields for consecutive rapid entries
      setAmountStr('')
      setTitle('')
      setNote('')
      setIsRecurring(false)

      setTimeout(() => {
        amountInputRef.current?.focus()
      }, 50)
    } catch (err: unknown) {
      console.error('Failed to save transaction:', err)
      setError('تعذر حفظ المعاملة. يرجى التحقق من البيانات والمحاولة مجدداً.')
    } finally {
      setSubmitting(false)
    }
  }

  const handleDelete = async () => {
    if (!user || !initialTransaction) return
    if (isLinkedToDebt) {
      setError('لا يمكن حذف معاملات الديون من هنا، يرجى حذف أو إدارة الدين من صفحة الديون.')
      return
    }
    setDeleting(true)
    try {
      await deleteTransaction(user.uid, initialTransaction.id)
      showToast('تم حذف المعاملة', 'info')
      onSuccess?.()
      onClose()
    } catch (err) {
      console.error('Failed to delete transaction:', err)
      setError('تعذر حذف المعاملة')
    } finally {
      setDeleting(false)
    }
  }

  const typeOptions: { value: TxType; label: string }[] = [
    { value: 'expense', label: 'مصروف' },
    { value: 'income', label: 'دخل' },
  ]

  return (
    <form onSubmit={handleSubmit} className="space-y-5">
      {/* Linked Debt Warning Banner */}
      {isLinkedToDebt && (
        <div className="p-3.5 rounded-2xl bg-amber-50 dark:bg-amber-950/40 border border-amber-300 dark:border-amber-800 text-amber-900 dark:text-amber-200 text-xs space-y-2">
          <div className="flex items-center gap-2 font-bold text-amber-800 dark:text-amber-300">
            <AlertCircle className="w-4 h-4 text-amber-600 dark:text-amber-400 shrink-0" />
            <span>معاملة متزامنة مع سجل الديون</span>
          </div>
          <p className="text-[11px] leading-relaxed text-amber-700/90 dark:text-amber-300/80 pe-1">
            هذه المعاملة مرتبطة بسجل الديون، لتعديل المبلغ أو حذفها يرجى إدارة الدفعة من صفحة الديون.
          </p>
          <Link
            to="/debts"
            onClick={onClose}
            className="inline-flex items-center gap-1.5 px-3.5 py-2 rounded-xl bg-amber-600 hover:bg-amber-700 text-white font-bold text-xs transition-colors shadow-xs"
          >
            <span>الانتقال إلى سجل الديون لتعديل أو حذف الدفعة</span>
            <ArrowLeft className="w-3.5 h-3.5 rtl:rotate-180" />
          </Link>
        </div>
      )}

      {/* Type Toggle: مصروف / دخل */}
      <div>
        <SegmentedControl<TxType>
          value={type}
          disabled={isLinkedToDebt}
          onChange={(newType) => {
            setType(newType)
            setCategoryId('')
            setError(null)
          }}
          options={typeOptions}
          className="w-full"
        />
      </div>

      {/* Big Amount Input */}
      <div>
        <label
          htmlFor="tx-amount"
          className="block text-xs font-bold text-slate-500 dark:text-slate-400 mb-1.5"
        >
          المبلغ
        </label>
        <div
          className={cn(
            'flex items-center justify-between px-4 py-2.5 bg-slate-50 dark:bg-slate-900/60 rounded-2xl border transition-all',
            error && amountStr && toFils(amountStr) <= 0
              ? 'border-red-500 ring-2 ring-red-500/20'
              : 'border-slate-200 dark:border-slate-700 focus-within:border-emerald-500 focus-within:ring-2 focus-within:ring-emerald-500/20',
            isLinkedToDebt && 'opacity-70 bg-slate-100 dark:bg-slate-800/80 cursor-not-allowed'
          )}
        >
          <input
            ref={amountInputRef}
            id="tx-amount"
            type="text"
            inputMode="decimal"
            readOnly={isLinkedToDebt}
            disabled={isLinkedToDebt}
            value={amountStr}
            onChange={(e) => {
              if (isLinkedToDebt) return
              setAmountStr(e.target.value)
              setError(null)
            }}
            placeholder={currencyConfig.decimals === 3 ? '0.000' : '0.00'}
            className={cn(
              'w-full bg-transparent text-2xl lg:text-3xl font-black text-slate-800 dark:text-slate-100 placeholder:text-slate-300 dark:placeholder:text-slate-600 outline-none',
              isLinkedToDebt && 'cursor-not-allowed'
            )}
            autoFocus={!isEditing && !isLinkedToDebt}
          />
          <span className="text-base lg:text-lg font-bold text-slate-400 dark:text-slate-500 select-none pe-1">
            {currencyConfig.symbol}
          </span>
        </div>
      </div>

      {/* Category Chips Grid */}
      <div>
        <div className="mb-2">
          <label className="block text-xs font-bold text-slate-500 dark:text-slate-400">
            التصنيف
          </label>
        </div>
        {categoriesLoading ? (
          <div className="py-4 text-center text-xs text-slate-400">جاري تحميل التصنيفات...</div>
        ) : availableCategories.length === 0 ? (
          <div className="py-3 text-center text-xs text-slate-400 bg-slate-50 dark:bg-slate-900 rounded-xl">
            لا توجد تصنيفات متاحة لهذا النوع
          </div>
        ) : (
          <div className="grid grid-cols-2 sm:grid-cols-3 gap-2 p-0.5">
            {availableCategories.map((cat) => {
              const isSelected = activeCategoryId === cat.id
              return (
                <button
                  key={cat.id}
                  type="button"
                  onClick={() => {
                    setCategoryId(cat.id)
                    setError(null)
                  }}
                  className={cn(
                    'flex items-center gap-2 p-2.5 rounded-xl border text-xs sm:text-sm font-bold transition-all text-start cursor-pointer select-none min-w-0',
                    isSelected
                      ? 'border-emerald-500 bg-emerald-50 text-emerald-900 dark:bg-emerald-950/60 dark:text-emerald-200 ring-2 ring-emerald-500/20 shadow-xs'
                      : 'border-slate-200 dark:border-slate-700/80 bg-white dark:bg-slate-800 text-slate-700 dark:text-slate-300 hover:bg-slate-50 dark:hover:bg-slate-700/50'
                  )}
                >
                  <span
                    className="w-7 h-7 rounded-lg flex items-center justify-center shrink-0 text-base"
                    style={{ backgroundColor: `${cat.color}25` }}
                  >
                    {cat.icon}
                  </span>
                  <span className="truncate flex-1 text-xs sm:text-sm font-semibold">{cat.name}</span>
                  {isSelected && (
                    <Check className="w-4 h-4 text-emerald-600 dark:text-emerald-400 shrink-0" />
                  )}
                </button>
              )
            })}
          </div>
        )}
      </div>

      {/* Date Input with Quick Select Buttons */}
      <div>
        <div className="flex items-center justify-between mb-1.5">
          <label
            htmlFor="tx-date"
            className="block text-xs font-bold text-slate-500 dark:text-slate-400"
          >
            التاريخ
          </label>
          <div className="flex items-center gap-1.5">
            <button
              type="button"
              onClick={() => {
                const today = new Date()
                setDateStr(format(today, 'yyyy-MM-dd'))
              }}
              className={cn(
                'px-2 py-0.5 rounded-lg text-xs font-semibold transition-colors cursor-pointer',
                dateStr === format(new Date(), 'yyyy-MM-dd')
                  ? 'bg-emerald-100 text-emerald-700 dark:bg-emerald-950/70 dark:text-emerald-300 font-bold'
                  : 'bg-slate-100 text-slate-600 hover:bg-slate-200/70 dark:bg-slate-800 dark:text-slate-400 dark:hover:bg-slate-700'
              )}
            >
              اليوم
            </button>
            <button
              type="button"
              onClick={() => {
                const yesterday = new Date()
                yesterday.setDate(yesterday.getDate() - 1)
                setDateStr(format(yesterday, 'yyyy-MM-dd'))
              }}
              className={cn(
                'px-2 py-0.5 rounded-lg text-xs font-semibold transition-colors cursor-pointer',
                (() => {
                  const y = new Date()
                  y.setDate(y.getDate() - 1)
                  return dateStr === format(y, 'yyyy-MM-dd')
                })()
                  ? 'bg-emerald-100 text-emerald-700 dark:bg-emerald-950/70 dark:text-emerald-300 font-bold'
                  : 'bg-slate-100 text-slate-600 hover:bg-slate-200/70 dark:bg-slate-800 dark:text-slate-400 dark:hover:bg-slate-700'
              )}
            >
              أمس
            </button>
          </div>
        </div>
        <Input
          id="tx-date"
          type="date"
          value={dateStr}
          onChange={(e) => setDateStr(e.target.value)}
          required
        />
      </div>

      {/* Description / Title Input */}
      <div>
        <Input
          type="text"
          label="الوصف / العنوان (اختياري)"
          placeholder="مثال: فاتورة كهرباء، مشتريات تموينية..."
          value={title}
          onChange={(e) => setTitle(e.target.value)}
          maxLength={100}
        />
      </div>

      {/* Note Input (Collapsible) */}
      {showNoteField ? (
        <div>
          <Input
            type="text"
            label="ملاحظة تفصيلية (اختياري)"
            placeholder="مثال: غداء مع الأصدقاء"
            value={note}
            onChange={(e) => setNote(e.target.value)}
            maxLength={500}
          />
        </div>
      ) : (
        <div className="-mt-2">
          <button
            type="button"
            onClick={() => setShowNoteField(true)}
            className="text-xs font-semibold text-emerald-600 dark:text-emerald-400 hover:underline inline-flex items-center gap-1 cursor-pointer"
          >
            <span>+ إضافة ملاحظة تفصيلية</span>
          </button>
        </div>
      )}

      {/* Recurring Option (only for new transactions) */}
      {!isEditing && (
        <div className="pt-2 border-t border-slate-100 dark:border-slate-800 space-y-3">
          <Toggle
            checked={isRecurring}
            onChange={setIsRecurring}
            label="جعلها معاملة متكررة"
            description="تكرار هذه المعاملة تلقائياً وفق دورية زمنية محددة"
          />

          {isRecurring && (
            <div className="space-y-1.5 pe-1">
              <label className="block text-xs font-bold text-slate-500 dark:text-slate-400">
                دورية التكرار
              </label>
              <SegmentedControl<Frequency>
                value={recurringFrequency}
                onChange={setRecurringFrequency}
                options={[
                  { value: 'weekly', label: 'أسبوعي' },
                  { value: 'monthly', label: 'شهري' },
                  { value: 'yearly', label: 'سنوي' },
                ]}
                className="w-full"
              />
            </div>
          )}
        </div>
      )}

      {/* Error message */}
      {error && (
        <div
          role="alert"
          className="flex items-center gap-2 p-3 text-start rounded-xl bg-red-50 dark:bg-red-950/40 border border-red-200 dark:border-red-800/50 text-red-600 dark:text-red-400 text-xs font-semibold"
        >
          <AlertCircle className="w-4 h-4 shrink-0" />
          <span>{error}</span>
        </div>
      )}

      {/* Delete confirmation if editing */}
      {isEditing && !isLinkedToDebt && showDeleteConfirm ? (
        <div className="p-3.5 rounded-2xl bg-red-50 dark:bg-red-950/40 border border-red-200 dark:border-red-900/60 space-y-2.5">
          <p className="text-xs font-bold text-red-800 dark:text-red-300">
            هل أنت متأكد من حذف هذه المعاملة نهائياً؟
          </p>
          <div className="flex items-center gap-2">
            <Button
              type="button"
              variant="danger"
              size="sm"
              onClick={handleDelete}
              disabled={deleting}
              className="flex-1"
            >
              {deleting ? 'جاري الحذف...' : 'نعم، احذف المعاملة'}
            </Button>
            <Button
              type="button"
              variant="outline"
              size="sm"
              onClick={() => setShowDeleteConfirm(false)}
              disabled={deleting}
            >
              تراجع
            </Button>
          </div>
        </div>
      ) : null}

      {/* Actions */}
      <div className="flex items-center justify-between gap-3 pt-2">
        <div className="flex items-center gap-2 flex-1 flex-wrap sm:flex-nowrap">
          <Button
            type="submit"
            variant="primary"
            disabled={submitting}
            className="flex-1"
          >
            {submitting
              ? 'جاري الحفظ...'
              : isEditing
                ? 'تعديل المعاملة'
                : duplicateFrom
                  ? 'حفظ النسخة'
                  : 'حفظ'}
          </Button>

          {!isEditing && (
            <Button
              type="button"
              variant="outline"
              onClick={handleSaveAndAddAnother}
              disabled={submitting}
              className="text-xs sm:text-sm font-semibold whitespace-nowrap"
            >
              حفظ وإضافة أخرى
            </Button>
          )}

          <Button
            type="button"
            variant="ghost"
            onClick={onClose}
            disabled={submitting}
          >
            إلغاء
          </Button>
        </div>

        {isEditing && !isLinkedToDebt && !showDeleteConfirm && (
          <Button
            type="button"
            variant="ghost"
            onClick={() => setShowDeleteConfirm(true)}
            aria-label="حذف المعاملة"
            className="text-red-600 hover:text-red-700 hover:bg-red-50 dark:text-red-400 dark:hover:bg-red-950/40 p-2.5"
          >
            <Trash2 className="w-4 h-4" />
          </Button>
        )}
      </div>
    </form>
  )
}

const DailyExpenseChart = React.lazy(() =>
  import('../components/charts/DailyExpenseChart').then((m) => ({ default: m.DailyExpenseChart }))
)
import { useAuth } from '../hooks/useAuth'
import { useSelectedMonth } from '../hooks/useSelectedMonth'
import { useTransactions } from '../hooks/useTransactions'
import { useCategories } from '../hooks/useCategories'
import { useRecurring } from '../hooks/useRecurring'
import { useDebts } from '../hooks/useDebts'
import { useUserSettings } from '../hooks/useUserSettings'
import { useCurrency } from '../hooks/useCurrency'
import { useTransactionModal } from '../hooks/useTransactionModal'
import { subscribePriorBalance } from '../services/transactions'
import { sumByType } from '../lib/aggregate'
import { calculateCategoryDistribution, calculateDailyExpenses } from '../lib/reports'
import { calculateDebtSummary } from '../lib/debts'
import { calculateForecast } from '../lib/forecast'
import { toWesternDigits, formatDayHeader } from '../lib/dates'
import { format as formatDate } from 'date-fns'
import { ar } from 'date-fns/locale'
import { cn } from '../lib/cn'

export const DashboardPage: React.FC = () => {
  const { user } = useAuth()
  const { range } = useSelectedMonth()
  const { data: rawTransactions, loading: txLoading } = useTransactions(range)
  const { data: categories } = useCategories()
  const { data: recurringList } = useRecurring()
  const { data: debts } = useDebts()
  const { data: settings } = useUserSettings()
  const { format, formatWithSign } = useCurrency()
  const { openAdd, openEdit, openDuplicate } = useTransactionModal()

  const [rolloverBalance, setRolloverBalance] = useState<number>(0)
  const isRolloverEnabled = Boolean(settings?.rolloverBalance)

  useEffect(() => {
    if (!user?.uid || !isRolloverEnabled) {
      return
    }
    const unsub = subscribePriorBalance(
      user.uid,
      range.start,
      (bal) => {
        setRolloverBalance(bal)
      },
      (err) => {
        console.error('Failed to subscribe prior balance:', err)
      }
    )
    return () => {
      unsub()
    }
  }, [user?.uid, isRolloverEnabled, range.start])

  const monthStartDay = settings?.monthStartDay ?? 1
  const effectiveRollover = isRolloverEnabled ? rolloverBalance : 0

  // Calculate monthly totals
  const sums = useMemo(() => sumByType(rawTransactions), [rawTransactions])
  const displayBalance = sums.balance + effectiveRollover

  // Calculate forecast & daily safe budget
  const forecast = useMemo(() => {
    return calculateForecast({
      transactions: rawTransactions,
      recurring: recurringList,
      debts,
      monthStartDay,
      periodRange: range,
      rolloverBalanceFils: effectiveRollover,
    })
  }, [rawTransactions, recurringList, debts, monthStartDay, range, effectiveRollover])

  // Calculate active debts summary
  const debtSummary = useMemo(() => calculateDebtSummary(debts), [debts])

  // Map categoryId to Category object
  const categoryMap = useMemo(() => {
    const map = new Map<string, (typeof categories)[0]>()
    for (const c of categories) {
      map.set(c.id, c)
    }
    return map
  }, [categories])

  // Category distribution for donut chart
  const categoryDistribution = useMemo(
    () => calculateCategoryDistribution(rawTransactions, categories, 'expense'),
    [rawTransactions, categories]
  )

  // Daily expenses for trend area chart
  const dailyExpenses = useMemo(
    () => calculateDailyExpenses(rawTransactions, range.start, range.end),
    [rawTransactions, range.start, range.end]
  )

  // Up to 5 most recent transactions
  const recentTransactions = useMemo(
    () => rawTransactions.slice(0, 5),
    [rawTransactions]
  )

  // Up to 5 upcoming active recurring items
  const upcomingRecurring = useMemo(() => {
    return recurringList
      .filter((r) => r.active)
      .sort((a, b) => {
        const dateA = a.nextDate instanceof Date ? a.nextDate : new Date(a.nextDate)
        const dateB = b.nextDate instanceof Date ? b.nextDate : new Date(b.nextDate)
        return dateA.getTime() - dateB.getTime()
      })
      .slice(0, 5)
  }, [recurringList])

  const today = useMemo(() => new Date(), [])

  // Guard against initial loading state where rawTransactions is [] and txLoading is true,
  // preventing calculation of deficit (danger/red flash) for fractions of a second before data arrives
  const isInitialOrLoading = txLoading && rawTransactions.length === 0

  // Dynamic Hero Card Palette based on financial health:
  // safe (excellent) -> Emerald / Green (lighter, vibrant, distinct)
  // warning (medium) -> Amber / Yellow (warm, bright, clear)
  // danger (bad) -> Rose / Red (striking, clear crimson)
  const heroTheme = useMemo(() => {
    let status: 'safe' | 'warning' | 'danger'
    if (isInitialOrLoading) {
      status = 'safe'
    } else if (displayBalance < 0 || forecast.freeBalance < 0) {
      status = 'danger'
    } else if (forecast.healthStatus === 'warning' || forecast.healthStatus === 'danger') {
      status = 'warning'
    } else {
      status = 'safe'
    }

    if (status === 'safe') {
      return {
        cardBg:
          'bg-gradient-to-br from-emerald-600 via-emerald-700 to-teal-800 border-emerald-400/40 shadow-xl shadow-emerald-900/15',
        walletBg: 'bg-white/20 text-white',
        badge: 'bg-white/20 text-white border-white/30 backdrop-blur-xs',
        badgeText: isInitialOrLoading
          ? 'جاري التحديث...'
          : 'فائض مالي',
        subBox: 'bg-white/10 hover:bg-white/15 border-white/20 text-white backdrop-blur-xs',
        subLabel: 'text-emerald-100',
        stripBg: 'bg-white/15 hover:bg-white/20 border-white/25 text-white backdrop-blur-xs',
        stripText: 'text-emerald-100',
      }
    }
    if (status === 'warning') {
      return {
        cardBg:
          'bg-gradient-to-br from-amber-600 via-amber-700 to-yellow-800 border-amber-400/40 shadow-xl shadow-amber-900/15',
        walletBg: 'bg-white/20 text-white',
        badge: 'bg-white/20 text-white border-white/30 backdrop-blur-xs',
        badgeText: 'انتبه للوتيرة',
        subBox: 'bg-white/10 hover:bg-white/15 border-white/20 text-white backdrop-blur-xs',
        subLabel: 'text-amber-100',
        stripBg: 'bg-white/15 hover:bg-white/20 border-white/25 text-white backdrop-blur-xs',
        stripText: 'text-amber-100',
      }
    }
    return {
      cardBg:
        'bg-gradient-to-br from-rose-600 via-red-700 to-rose-800 border-rose-400/40 shadow-xl shadow-rose-900/20',
      walletBg: 'bg-white/20 text-white',
      badge: 'bg-white/20 text-white border-white/30 backdrop-blur-xs',
      badgeText: displayBalance < 0 ? 'عجز مالي' : 'التزامات تفوق الرصيد',
      subBox: 'bg-white/10 hover:bg-white/15 border-white/20 text-white backdrop-blur-xs',
      subLabel: 'text-rose-100',
      stripBg: 'bg-white/15 hover:bg-white/20 border-white/25 text-white backdrop-blur-xs',
      stripText: 'text-rose-100',
    }
  }, [isInitialOrLoading, forecast.healthStatus, forecast.freeBalance, displayBalance])

  return (
    <div className="space-y-6 animate-slide-up">
      {/* Header with Title and Desktop-only Quick Add */}
      <PageHeader
        title="لوحة التحكم"
        subtitle="نظرة شاملة وسريعة على وضعك المالي لهذا الشهر"
        actions={
          <Button
            variant="primary"
            size="md"
            onClick={() => openAdd('expense')}
            className="hidden lg:inline-flex gap-1.5 shadow-xs font-semibold press-scale"
          >
            <Plus className="w-4 h-4" />
            <span>إضافة معاملة</span>
          </Button>
        }
      />

      {/* 1. Hero Card: Clean, Vibrant, High-Contrast Cash Balance */}
      {isInitialOrLoading ? (
        <HeroCardSkeleton />
      ) : (
        <div
          className={cn(
            'text-white rounded-2xl sm:rounded-3xl p-5 sm:p-6 shadow-md border space-y-4 transition-colors duration-300',
            heroTheme.cardBg
          )}
        >
          <div className="flex items-center justify-between">
            <div className="flex items-center gap-2.5">
              <div
                className={cn(
                  'w-8 h-8 rounded-xl flex items-center justify-center',
                  heroTheme.walletBg
                )}
              >
                <Wallet className="w-4 h-4 stroke-[2.2]" />
              </div>
              <span className="text-xs sm:text-sm font-semibold text-white/90">
                {isRolloverEnabled ? 'الرصيد المتاح (شامل المرحّل)' : 'الرصيد المتاح (لهذا الشهر)'}
              </span>
            </div>
            <span
              className={cn(
                'text-xs font-black px-2.5 py-0.5 rounded-full select-none border',
                heroTheme.badge
              )}
            >
              {heroTheme.badgeText}
            </span>
          </div>

          <div className="text-3xl sm:text-4xl lg:text-5xl font-black tracking-tight select-none">
            {format(displayBalance)}
          </div>

          {/* Rollover Breakdown Subline */}
          {isRolloverEnabled && (
            <div className="flex flex-wrap items-center gap-x-2.5 gap-y-1.5 text-xs sm:text-sm font-medium text-white/95 -mt-1 pb-1">
              <div className="inline-flex items-center gap-1.5 px-2.5 py-1 rounded-xl bg-white/15 backdrop-blur-xs border border-white/20">
                <span className="text-white/80 text-[11px] sm:text-xs">صافي هذا الشهر:</span>
                <span className="font-black text-white">
                  {sums.balance > 0
                    ? `+ ${format(sums.balance)}`
                    : sums.balance < 0
                      ? `- ${format(Math.abs(sums.balance))}`
                      : format(0)}
                </span>
              </div>
              <div className="inline-flex items-center gap-1.5 px-2.5 py-1 rounded-xl bg-white/15 backdrop-blur-xs border border-white/20">
                <span className="text-white/80 text-[11px] sm:text-xs">الرصيد المرحّل:</span>
                <span className="font-black text-white">
                  {effectiveRollover > 0
                    ? `+ ${format(effectiveRollover)}`
                    : effectiveRollover < 0
                      ? `- ${format(Math.abs(effectiveRollover))}`
                      : format(0)}
                </span>
              </div>
            </div>
          )}

          {/* Total Income & Total Expenses Row */}
          <div className="grid grid-cols-2 gap-3 pt-3 border-t border-white/15">
            <div
              className={cn(
                'flex items-center gap-2.5 p-2.5 rounded-xl border',
                heroTheme.subBox
              )}
            >
              <div className="w-8 h-8 rounded-lg bg-white/20 flex items-center justify-center text-white shrink-0 font-bold">
                <TrendingUp className="w-4 h-4 stroke-[2.5]" />
              </div>
              <div className="min-w-0">
                <div className={cn('text-[11px] font-medium', heroTheme.subLabel)}>
                  إجمالي المداخيل
                </div>
                <div className="text-sm sm:text-base font-black truncate text-white">
                  {format(sums.income)}
                </div>
              </div>
            </div>

            <div
              className={cn(
                'flex items-center gap-2.5 p-2.5 rounded-xl border',
                heroTheme.subBox
              )}
            >
              <div className="w-8 h-8 rounded-lg bg-white/20 flex items-center justify-center text-white shrink-0 font-bold">
                <TrendingDown className="w-4 h-4 stroke-[2.5]" />
              </div>
              <div className="min-w-0">
                <div className={cn('text-[11px] font-medium', heroTheme.subLabel)}>
                  إجمالي المصاريف
                </div>
                <div className="text-sm sm:text-base font-black truncate text-white">
                  {format(sums.expense)}
                </div>
              </div>
            </div>
          </div>

          {/* Daily Safe Budget Interactive Strip */}
          {forecast.isCurrentMonth && (
            <Link
              to="/reports"
              className={cn(
                'press-scale flex items-center justify-between p-2.5 rounded-xl border text-xs transition-colors group cursor-pointer',
                heroTheme.stripBg
              )}
              title="عرض التفاصيل في التقارير"
            >
              <div className="flex items-center gap-2 text-white font-medium">
                <div className="w-6 h-6 rounded-lg bg-white/20 text-white flex items-center justify-center">
                  <ShieldCheck className="w-3.5 h-3.5" />
                </div>
                <span>ميزانية الأمان اليومي:</span>
              </div>
              <div className="flex items-center gap-1.5 text-end">
                <span className="font-black text-white text-sm sm:text-base">
                  {format(forecast.dailySafeFils)}
                </span>
                <span className={cn('text-[11px] font-medium', heroTheme.stripText)}>
                  / اليوم
                </span>
                <ChevronLeft className="w-3.5 h-3.5 text-white/80 group-hover:-translate-x-0.5 transition-transform" />
              </div>
            </Link>
          )}
        </div>
      )}

      {/* Quick Feature Access Hub (Mobile only, hidden on lg desktop) */}
      <QuickFeatureHub />

      {/* 2. Core Activity: Recent Transactions (col 7) & Debts Summary + Upcoming Recurring (col 5) */}
      <div className="grid grid-cols-1 lg:grid-cols-12 gap-5 items-start">
        {/* Recent Transactions: lg:col-span-7 */}
        <div className="lg:col-span-7">
          <Card className="p-4 sm:p-5">
            <div className="flex items-center justify-between mb-3 pb-2 border-b border-slate-100 dark:border-slate-700/60">
              <div className="flex items-center gap-2">
                <Receipt className="w-4 h-4 text-emerald-500" />
                <h2 className="text-sm sm:text-base font-extrabold text-slate-800 dark:text-slate-100">
                  آخر المعاملات
                </h2>
              </div>
              <Link
                to="/transactions"
                className="text-xs font-bold text-emerald-600 dark:text-emerald-400 hover:underline flex items-center gap-0.5"
              >
                <span>عرض الكل</span>
                <ChevronLeft className="w-3.5 h-3.5" />
              </Link>
            </div>

            {isInitialOrLoading ? (
              <div className="space-y-2.5">
                {[1, 2, 3, 4].map((i) => (
                  <div
                    key={i}
                    className="flex items-center justify-between p-2.5 rounded-xl bg-slate-50/70 dark:bg-slate-750/40 animate-pulse"
                  >
                    <div className="flex items-center gap-3 min-w-0">
                      <div className="w-9 h-9 rounded-xl bg-slate-200 dark:bg-slate-700 shrink-0" />
                      <div className="space-y-1.5">
                        <div className="w-28 h-3.5 rounded bg-slate-200 dark:bg-slate-700" />
                        <div className="w-16 h-2.5 rounded bg-slate-100 dark:bg-slate-750" />
                      </div>
                    </div>
                    <div className="w-16 h-4 rounded bg-slate-200 dark:bg-slate-700 shrink-0" />
                  </div>
                ))}
              </div>
            ) : recentTransactions.length === 0 ? (
              <EmptyState
                icon={<Receipt className="w-6 h-6" />}
                title="لا توجد معاملات مسجلة"
                description="أضف معاملتك الأولى لهذا الشهر للبدء بتتبع مصاريفك ومداخيلك."
                action={
                  <Button
                    variant="primary"
                    size="sm"
                    onClick={() => openAdd('expense')}
                    className="gap-1 mt-1"
                  >
                    <Plus className="w-4 h-4" />
                    <span>إضافة معاملة</span>
                  </Button>
                }
              />
            ) : (
              <div className="space-y-1.5">
                {recentTransactions.map((tx) => {
                  const cat = categoryMap.get(tx.categoryId)
                  const isExpense = tx.type === 'expense'
                  const txDate = tx.date instanceof Date ? tx.date : new Date(tx.date)
                  const dateHeader = formatDayHeader(txDate, today)

                  return (
                    <div
                      key={tx.id}
                      onClick={() => openEdit(tx)}
                      className="press-scale flex items-center justify-between p-2.5 rounded-xl hover:bg-slate-50 dark:hover:bg-slate-750 transition-colors cursor-pointer group"
                    >
                      <div className="flex items-center gap-3 min-w-0">
                        <span
                          className="w-9 h-9 rounded-xl flex items-center justify-center shrink-0 text-base shadow-xs"
                          style={{
                            backgroundColor: cat ? `${cat.color}25` : '#94A3B825',
                          }}
                        >
                          {cat ? cat.icon : '📁'}
                        </span>
                        <div className="min-w-0">
                          <div className="font-bold text-xs sm:text-sm text-slate-800 dark:text-slate-100 truncate group-hover:text-emerald-600 dark:group-hover:text-emerald-400 transition-colors">
                            {tx.title || tx.note || (cat ? cat.name : 'معاملة')}
                          </div>
                          <div className="flex items-center gap-1.5 text-[11px] text-slate-400 dark:text-slate-500">
                            {cat && (
                              <>
                                <span className="truncate">{cat.name}</span>
                                <span>•</span>
                              </>
                            )}
                            <span className="truncate">{dateHeader}</span>
                          </div>
                        </div>
                      </div>

                      <div className="flex items-center gap-1.5 shrink-0 ps-3">
                        <button
                          type="button"
                          onClick={(e) => {
                            e.stopPropagation()
                            openDuplicate(tx)
                          }}
                          className="hidden sm:inline-flex opacity-0 group-hover:opacity-100 p-1 rounded-lg text-slate-400 hover:text-emerald-600 hover:bg-emerald-50 dark:hover:text-emerald-400 dark:hover:bg-emerald-950/40 transition-all cursor-pointer"
                          aria-label="تكرار المعاملة"
                          title="تكرار المعاملة"
                        >
                          <Copy className="w-3.5 h-3.5" />
                        </button>
                        <div
                          className={cn(
                            'text-xs sm:text-sm font-extrabold select-none ps-0.5',
                            isExpense
                              ? 'text-red-500 dark:text-red-400'
                              : 'text-emerald-600 dark:text-emerald-400'
                          )}
                        >
                          {formatWithSign(tx.amountFils, tx.type)}
                        </div>
                      </div>
                    </div>
                  )
                })}
              </div>
            )}
          </Card>
        </div>

        {/* Debts Summary & Upcoming Recurring: lg:col-span-5 */}
        <div className="lg:col-span-5 space-y-5">
          {/* Active Debts Summary */}
          <Card className="p-4 sm:p-5 border border-slate-200/80 dark:border-slate-700/60 bg-white dark:bg-slate-800 rounded-2xl">
            <div className="flex items-center justify-between mb-3 pb-2 border-b border-slate-100 dark:border-slate-700/60">
              <div className="flex items-center gap-2">
                <HandCoins className="w-4 h-4 text-emerald-500" />
                <h2 className="text-sm sm:text-base font-extrabold text-slate-800 dark:text-slate-100">
                  ملخص الديون والالتزامات
                </h2>
              </div>
              <Link
                to="/debts"
                className="text-xs font-bold text-emerald-600 dark:text-emerald-400 hover:underline flex items-center gap-0.5"
              >
                <span>إدارة الديون</span>
                <ChevronLeft className="w-3.5 h-3.5" />
              </Link>
            </div>

            <div className="grid grid-cols-1 sm:grid-cols-3 lg:grid-cols-1 gap-2.5">
              {/* Lent to others (مستحقات لك) */}
              <div className="flex items-center justify-between p-2.5 rounded-xl bg-emerald-50/60 dark:bg-emerald-950/20 border border-emerald-100 dark:border-emerald-900/30">
                <div className="flex items-center gap-2">
                  <div className="w-7 h-7 rounded-lg bg-emerald-100 dark:bg-emerald-900/40 text-emerald-700 dark:text-emerald-300 flex items-center justify-center font-bold text-xs">
                    لي
                  </div>
                  <div>
                    <div className="text-xs font-bold text-slate-700 dark:text-slate-200">
                      مستحقات لك
                    </div>
                    <div className="text-[10px] text-slate-400">
                      {debtSummary.lentActiveCount} ديون نشطة
                    </div>
                  </div>
                </div>
                <div className="text-xs sm:text-sm font-black text-emerald-600 dark:text-emerald-400 select-none">
                  {format(debtSummary.lentFils)}
                </div>
              </div>

              {/* Borrowed from others (التزامات عليك) */}
              <div className="flex items-center justify-between p-2.5 rounded-xl bg-rose-50/60 dark:bg-rose-950/20 border border-rose-100 dark:border-rose-900/30">
                <div className="flex items-center gap-2">
                  <div className="w-7 h-7 rounded-lg bg-rose-100 dark:bg-rose-900/40 text-rose-700 dark:text-rose-300 flex items-center justify-center font-bold text-xs">
                    عليّ
                  </div>
                  <div>
                    <div className="text-xs font-bold text-slate-700 dark:text-slate-200">
                      التزامات عليك
                    </div>
                    <div className="text-[10px] text-slate-400">
                      {debtSummary.borrowedActiveCount} ديون نشطة
                    </div>
                  </div>
                </div>
                <div className="text-xs sm:text-sm font-black text-rose-600 dark:text-rose-400 select-none">
                  {format(debtSummary.borrowedFils)}
                </div>
              </div>

              {/* Net difference (الصافي) */}
              <div className="flex items-center justify-between p-2.5 rounded-xl bg-slate-50 dark:bg-slate-750/60 border border-slate-100 dark:border-slate-700/40">
                <div className="flex items-center gap-2">
                  <div className="w-7 h-7 rounded-lg bg-slate-200 dark:bg-slate-700 text-slate-700 dark:text-slate-200 flex items-center justify-center font-bold text-xs">
                    <Scale className="w-3.5 h-3.5" />
                  </div>
                  <div>
                    <div className="text-xs font-bold text-slate-700 dark:text-slate-200">
                      صافي المركز
                    </div>
                    <div className="text-[10px] text-slate-400">
                      {debtSummary.netFils >= 0 ? 'فائض لصالحك' : 'عجز عليك'}
                    </div>
                  </div>
                </div>
                <div
                  className={cn(
                    'text-xs sm:text-sm font-black select-none',
                    debtSummary.netFils >= 0
                      ? 'text-emerald-600 dark:text-emerald-400'
                      : 'text-rose-600 dark:text-rose-400'
                  )}
                >
                  {debtSummary.netFils >= 0 ? '+' : ''}
                  {format(debtSummary.netFils)}
                </div>
              </div>
            </div>
          </Card>

          {/* Upcoming Recurring */}
          <Card className="p-4 sm:p-5">
            <div className="flex items-center justify-between mb-3 pb-2 border-b border-slate-100 dark:border-slate-700/60">
              <div className="flex items-center gap-2">
                <Calendar className="w-4 h-4 text-emerald-500" />
                <h2 className="text-sm sm:text-base font-extrabold text-slate-800 dark:text-slate-100">
                  المتكررات القادمة
                </h2>
              </div>
              <Link
                to="/recurring"
                className="text-xs font-bold text-emerald-600 dark:text-emerald-400 hover:underline flex items-center gap-0.5"
              >
                <span>إدارة المتكررة</span>
                <ChevronLeft className="w-3.5 h-3.5" />
              </Link>
            </div>

            {upcomingRecurring.length === 0 ? (
              <div className="py-6 text-center text-xs text-slate-400 dark:text-slate-500">
                <p className="font-semibold mb-2">لا توجد معاملات متكررة نشطة</p>
                <Link
                  to="/recurring"
                  className="inline-flex items-center gap-1 text-emerald-600 dark:text-emerald-400 hover:underline font-bold"
                >
                  <span>ضبط معاملة متكررة جديدة</span>
                  <ArrowUpRight className="w-3.5 h-3.5" />
                </Link>
              </div>
            ) : (
              <div className="space-y-1.5">
                {upcomingRecurring.map((item) => {
                  const cat = categoryMap.get(item.categoryId)
                  const isExpense = item.type === 'expense'
                  const nextDate =
                    item.nextDate instanceof Date
                      ? item.nextDate
                      : new Date(item.nextDate)

                    formatDate(nextDate, 'd MMMM', { locale: ar })
                  )

                  return (

                          style={{
                            backgroundColor: cat ? `${cat.color}25` : '#94A3B825',
                          }}
                        >
                          {cat ? cat.icon : '🔄'}
                        </span>
                        <div className="min-w-0">
                          <div className="font-bold text-xs text-slate-800 dark:text-slate-100 truncate">
                            {item.title || item.note || (cat ? cat.name : 'معاملة متكررة')}
                          </div>
                          <div className="text-[10px] text-slate-400 dark:text-slate-500">
                            {nextDateFormatted}
                          </div>
                        </div>
                      </div>

                      <div className="text-end shrink-0 ps-2">
                        <div
                          className={cn(
                            'text-xs font-extrabold select-none',
                            isExpense
                              ? 'text-red-500 dark:text-red-400'
                              : 'text-emerald-600 dark:text-emerald-400'
                          )}
                        >
                          {formatWithSign(item.amountFils, item.type)}
                        </div>
                      </div>
                    </div>
                  )
                })}
              </div>
            )}
          </Card>
        </div>
      </div>

      {/* 3. Analytical Charts at the Bottom */}
      <div className="grid grid-cols-1 lg:grid-cols-2 gap-5">
        {/* Category Pie Chart Card */}

              المصاريف حسب التصنيف
            </h2>
            <span className="text-xs font-semibold text-slate-400 dark:text-slate-500">
              {categoryDistribution.length} تصنيفات
            </span>
          </div>
          <div className="flex-1 flex items-center justify-center">
            <React.Suspense
              fallback={
                <div className="h-56 w-full flex items-center justify-center">
                  <div className="w-32 h-32 rounded-full border-8 border-slate-200/80 dark:border-slate-700/60 animate-pulse" />
                </div>
              }
            >
              <CategoryPieChart data={categoryDistribution} className="w-full" />
            </React.Suspense>
          </div>
        </Card>

        {/* Daily Expenses Chart Card */}
        <Card className="p-4 sm:p-5 flex flex-col justify-between">
          <div className="flex items-center justify-between mb-4">
            <h2 className="text-sm sm:text-base font-extrabold text-slate-800 dark:text-slate-100">
              المصاريف اليومية
            </h2>
            <span className="text-xs font-semibold text-slate-400 dark:text-slate-500">
              حركة الصرف اليومية
            </span>
          </div>
          <div className="flex-1 flex items-center justify-center">
            <React.Suspense
              fallback={
                <div className="h-56 w-full flex items-end justify-between gap-2 px-4 pb-4 animate-pulse">
                  {[40, 65, 30, 80, 50, 95, 45, 70, 60, 85].map((height, idx) => (

                      key={idx}
                      className="w-full bg-slate-200/80 dark:bg-slate-700/60 rounded-t-lg"
                      style={{ height: `${height}%` }}
                    />
                  ))}
                </div>
              }
            >
              <DailyExpenseChart data={dailyExpenses} className="w-full" />

}

  const deferredQuery = useDeferredValue(query)
  const [filterType, setFilterType] = useState<TxType | 'all'>('all')

    query.trim() || filterType !== 'all' || selectedCategory !== 'all'
  )

    setQuery('')
    setFilterType('all')
    setSelectedCategory('all')
  }

  const categoryMap = useMemo(() => {
    const map = new Map<string, (typeof categories)[0]>()
    for (const c of categories) {
      map.set(c.id, c)
    }
    return map
  }, [categories])

  // Track category usage from transactions in memory (read-only calculation during render)

    const txMap = buildUsageFromTransactions(rawTransactions)
    return mergeUsageMaps(txMap, getCategoryUsageMap())
  }, [rawTransactions])

  // Save merged category usage in localStorage outside render in an effect
  useEffect(() => {
    if (rawTransactions.length > 0) {
      saveCategoryUsageMap(usageMap)
    }
  }, [rawTransactions.length, usageMap])

  // Relevant categories for horizontal chips (sorted by usage frequency)

    const filtered = categories.filter(
      (c) => (filterType === 'all' ? true : c.type === filterType) && !c.archived
    )
    return sortCategoriesByUsage(filtered, usageMap)
  }, [categories, filterType, usageMap])

  // Filtered transactions (using deferredQuery to prevent keyboard lag on mobile)
  const filteredTransactions = useMemo(() => {

  }, [rawTransactions, deferredQuery, filterType, selectedCategory])

  downloadBackupFile,
  validateBackup,
  restoreBackup,
  type BackupValidationResult,
} from '../lib/backupRestore'
import { PRESET_CURRENCIES, getCurrencyConfig } from '../lib/money'
import { cn } from '../lib/cn'
import type { ThemePreference } from '../types'

export const SettingsPage: React.FC = () => {
  const { theme, setTheme, resolvedTheme } = useTheme()
  const { user } = useAuth()
  const { data: settings, updateSettings } = useUserSettings()
  const { data: categories } = useCategories()
  const { showToast } = useToast()
  const { isInstallable, isInstalled, promptInstall } = usePWAInstall()

  const [selectedDay, setSelectedDay] = useState<string | null>(null)
  const [loggingOut, setLoggingOut] = useState(false)
  const [exportingTxs, setExportingTxs] = useState(false)
  const [exportingDebts, setExportingDebts] = useState(false)
  const [exportingJson, setExportingJson] = useState(false)
  const [restoringJson, setRestoringJson] = useState(false)
  const [restoreValidation, setRestoreValidation] = useState<BackupValidationResult | null>(null)
  const fileInputRef = useRef<HTMLInputElement>(null)

  const activeCurrency = settings?.currency || 'JOD'
  const isPreset = Boolean(PRESET_CURRENCIES[activeCurrency])
  const [isCustomCurrency, setIsCustomCurrency] = useState(!isPreset && activeCurrency !== 'JOD')
  const [customCurrencyInput, setCustomCurrencyInput] = useState(isPreset ? '' : activeCurrency)
  const [savingCurrency, setSavingCurrency] = useState(false)

  // Track previous settings to adjust state during render when settings load or change from Firestore
  const [prevCurrency, setPrevCurrency] = useState(settings?.currency)
  if (settings?.currency !== prevCurrency) {
    setPrevCurrency(settings?.currency)
    const isCustom = settings?.currency ? !PRESET_CURRENCIES[settings.currency] : false
    setIsCustomCurrency(isCustom)
    setCustomCurrencyInput(isCustom ? settings?.currency || '' : '')
  }

  const [prevMonthStartDay, setPrevMonthStartDay] = useState(settings?.monthStartDay)
  if (settings?.monthStartDay !== prevMonthStartDay) {
    setPrevMonthStartDay(settings?.monthStartDay)
    if (settings?.monthStartDay !== undefined) {
      setSelectedDay(String(settings.monthStartDay))
    }
  }

  const activeConfig = getCurrencyConfig(activeCurrency)

  const handleSelectPreset = async (code: string) => {
    setIsCustomCurrency(false)
    try {
      await updateSettings({ currency: code })
      showToast(`تم تغيير العملة إلى ${PRESET_CURRENCIES[code]?.name || code} بنجاح`, 'success')
    } catch (err) {
      console.error('Failed to update currency:', err)
      showToast('تعذر تغيير العملة', 'error')
    }
  }

  const handleSaveCustomCurrency = async () => {
    const trimmed = customCurrencyInput.trim()
    if (!trimmed) return
    setSavingCurrency(true)
    try {
      await updateSettings({ currency: trimmed })
      showToast(`تم تعيين العملة المخصصة: ${trimmed}`, 'success')
    } catch (err) {
      console.error('Failed to save custom currency:', err)
      showToast('تعذر حفظ العملة المخصصة', 'error')
    } finally {
      setSavingCurrency(false)
    }
  }

  const currentStartDay =
    selectedDay ?? (settings?.monthStartDay ? String(settings.monthStartDay) : '1')

  const themeOptions: { value: ThemePreference; label: string; icon: React.ReactNode }[] = [
    { value: 'system', label: 'تلقائي', icon: <Laptop className="w-4 h-4" /> },
    { value: 'light', label: 'فاتح', icon: <Sun className="w-4 h-4" /> },
    { value: 'dark', label: 'داكن', icon: <Moon className="w-4 h-4" /> },
  ]

  const dayOptions = Array.from({ length: 28 }, (_, i) => ({
    value: String(i + 1),
    label: `يوم ${i + 1} من كل شهر`,
  }))

  const handleThemeChange = async (newTheme: ThemePreference) => {
    setTheme(newTheme)
    try {
      await updateSettings({ theme: newTheme })
    } catch (err) {
      console.error('Failed to persist theme:', err)
    }
  }

  const handleStartDayChange = async (e: React.ChangeEvent<HTMLSelectElement>) => {
    const val = e.target.value
    setSelectedDay(val)
    try {
      await updateSettings({ monthStartDay: parseInt(val, 10) })
    } catch (err) {
      console.error('Failed to persist month start day:', err)
    }
  }

  const handleSignOut = async () => {
    setLoggingOut(true)
    try {
      await signOutUser()
    } catch (err) {
      console.error('Failed to sign out:', err)
    } finally {
      setLoggingOut(false)
    }
  }

  const handleExportTransactions = async () => {
    if (!user) return
    setExportingTxs(true)
    try {
      const txs = await getAllTransactions(user.uid)
      if (txs.length === 0) {
        showToast('لا توجد أي معاملات لتصديرها', 'info')
        return
      }
      exportTransactionsToCsv(txs, categories, {
        symbol: activeConfig.symbol,
        decimals: activeConfig.decimals,
      })
      showToast('تم تصدير المعاملات بنجاح', 'success')
    } catch (err) {
      console.error('Failed to export transactions:', err)
      showToast('تعذر تصدير المعاملات', 'error')
    } finally {
      setExportingTxs(false)
    }
  }

  const handleExportDebts = async () => {
    if (!user) return
    setExportingDebts(true)
    try {
      const debts = await getAllDebts(user.uid)
      if (debts.length === 0) {
        showToast('لا توجد أي ديون لتصديرها', 'info')
        return
      }
      exportDebtsToCsv(debts, {
        symbol: activeConfig.symbol,
        decimals: activeConfig.decimals,
      })
      showToast('تم تصدير سجل الديون بنجاح', 'success')
    } catch (err) {
      console.error('Failed to export debts:', err)
      showToast('تعذر تصدير سجل الديون', 'error')
    } finally {
      setExportingDebts(false)
    }
  }

  const handleToggleRollover = async (checked: boolean) => {
    try {
      await updateSettings({ rolloverBalance: checked })
      showToast(
        checked
          ? 'تم تفعيل ترحيل الرصيد المتبقي تلقائياً'
          : 'تم إيقاف ترحيل الرصيد، يعتمد التطبيق الآن على كل شهر منفصلاً',
        'success'
      )
    } catch (err) {
      console.error('Failed to toggle rollover balance:', err)
      showToast('تعذر تحديث إعدادات ترحيل الرصيد', 'error')
    }
  }

  const handleDownloadJsonBackup = async () => {
    if (!user) return
    setExportingJson(true)
    try {
      const backup = await createFullBackup(user.uid)
      downloadBackupFile(backup)
      showToast('تم تنزيل النسخة الاحتياطية الشاملة بنجاح', 'success')
    } catch (err) {
      console.error('Failed to export full backup:', err)
      showToast('تعذر إنشاء النسخة الاحتياطية', 'error')
    } finally {
      setExportingJson(false)
    }
  }

  const handleTriggerFileInput = () => {
    if (fileInputRef.current) {
      fileInputRef.current.value = ''
      fileInputRef.current.click()
    }
  }

  const handleFileSelected = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0]
    if (!file) return

    const reader = new FileReader()
    reader.onload = (event) => {
      const text = event.target?.result
      if (typeof text !== 'string') return
      const res = validateBackup(text)
      if (!res.valid) {
        showToast(res.error || 'ملف غير صالح', 'error')
        return
      }
      setRestoreValidation(res)
    }
    reader.onerror = () => {
      showToast('تعذر قراءة الملف المحدد', 'error')
    }
    reader.readAsText(file, 'utf-8')
  }

  const handleConfirmRestore = async () => {
    if (!user || !restoreValidation?.data) return
    setRestoringJson(true)
    try {
      await restoreBackup(user.uid, restoreValidation.data)
      showToast('تمت استعادة النسخة الاحتياطية بنجاح!', 'success')
      setRestoreValidation(null)
    } catch (err) {
      console.error('Failed to restore backup:', err)
      showToast('تعذر استعادة النسخة الاحتياطية', 'error')
    } finally {
      setRestoringJson(false)
    }
  }

  return (
    <div className="space-y-6 max-w-4xl mx-auto animate-slide-up">
      <PageHeader
        title="الإعدادات"
        subtitle="تفضيلات المظهر، بداية الشهر المالي، وإدارة الحساب"
      />

      {/* Theme Settings Card */}
      <Card
        title="المظهر والألوان"
        subtitle="اختر المظهر المفضل للتطبيق أو دعه يتبع إعدادات جهازك"
      >
        <div className="space-y-4 pt-2">
          <SegmentedControl<ThemePreference>
            value={theme}
            onChange={handleThemeChange}
            options={themeOptions}
            className="w-full sm:w-auto"
          />

          <p className="text-xs text-slate-500 dark:text-slate-400">
            الوضع الحالي النشط:{' '}
            <span className="font-semibold text-emerald-600 dark:text-emerald-400">
              {resolvedTheme === 'dark' ? 'داكن' : 'فاتح'}
            </span>{' '}
            {theme === 'system' && '(تلقائي حسب النظام)'}
          </p>
        </div>
      </Card>

      {/* Financial Month Start Day & Rollover */}
      <Card
        title="إعدادات الدورة المالية"
        subtitle="حدد يوم بداية الدورة المالية وإمكانية ترحيل الرصيد التراكمي من الأشهر السابقة"
      >
        <div className="pt-2 space-y-5">
          <div className="max-w-xs">
            <Select
              label="يوم بداية الدورة المالية"
              value={currentStartDay}
              onChange={handleStartDayChange}
              options={dayOptions}
              helperText="الافتراضي هو اليوم الأول من كل شهر (1 إلى 28)."
            />
          </div>

          <div className="pt-4 border-t border-slate-100 dark:border-slate-800 flex items-start justify-between gap-4">
            <div className="space-y-1">
              <label htmlFor="rollover-toggle" className="text-sm font-bold text-slate-800 dark:text-slate-200 cursor-pointer">
                ترحيل الرصيد المتبقي من الأشهر السابقة تلقائياً
              </label>
              <p className="text-xs text-slate-500 dark:text-slate-400 leading-relaxed">
                عند التفعيل، يُضاف الفائض أو العجز المتراكم من الأشهر السابقة إلى رصيد الشهر الحالي، مما يمنحك رصيداً تراكمياً حقيقياً وتفادي العجز الوهمي في بداية الشهر.
              </p>
            </div>
            <Toggle
              id="rollover-toggle"
              checked={Boolean(settings?.rolloverBalance)}
              onChange={handleToggleRollover}
            />
          </div>
        </div>
      </Card>

      {/* Currency Selection Card */}
      <Card
        title="العملة الرسمية"
        subtitle="اختر العملة المعتمدة في حساباتك وشاشات التطبيق"
      >
        <div className="space-y-4 pt-2">
          {/* Preset Buttons Grid */}
          <div className="grid grid-cols-2 sm:grid-cols-3 gap-2.5">
            {Object.values(PRESET_CURRENCIES).map((curr) => {
              const isSelected = !isCustomCurrency && activeCurrency === curr.code
              return (
                <button
                  key={curr.code}
                  type="button"
                  onClick={() => handleSelectPreset(curr.code)}
                  className={cn(
                    'flex items-center gap-2.5 p-3 rounded-xl border text-start transition-all cursor-pointer',
                    isSelected
                      ? 'border-emerald-500 bg-emerald-50/70 dark:bg-emerald-950/40 text-emerald-900 dark:text-emerald-200 ring-2 ring-emerald-500/20'
                      : 'border-slate-200 dark:border-slate-700 bg-slate-50/50 dark:bg-slate-800/50 hover:bg-slate-100 dark:hover:bg-slate-800 text-slate-700 dark:text-slate-300'
                  )}
                >
                  <span className="text-xl select-none">{curr.flag}</span>
                  <div className="min-w-0 flex-1">
                    <div className="text-xs font-bold truncate">{curr.name}</div>
                    <div className="text-[11px] text-slate-400 dark:text-slate-500 font-medium">
                      {curr.code} ({curr.symbol})
                    </div>
                  </div>
                </button>
              )
            })}

            {/* Custom Currency Option */}
            <button
              type="button"
              onClick={() => setIsCustomCurrency(true)}
              className={cn(
                'flex items-center gap-2.5 p-3 rounded-xl border text-start transition-all cursor-pointer',
                isCustomCurrency
                  ? 'border-emerald-500 bg-emerald-50/70 dark:bg-emerald-950/40 text-emerald-900 dark:text-emerald-200 ring-2 ring-emerald-500/20'
                  : 'border-slate-200 dark:border-slate-700 bg-slate-50/50 dark:bg-slate-800/50 hover:bg-slate-100 dark:hover:bg-slate-800 text-slate-700 dark:text-slate-300'
              )}
            >
              <span className="text-xl select-none">✏️</span>
              <div className="min-w-0 flex-1">
                <div className="text-xs font-bold truncate">عملة أخرى</div>
                <div className="text-[11px] text-slate-400 dark:text-slate-500 font-medium truncate">
                  {isCustomCurrency && customCurrencyInput ? customCurrencyInput : 'تحديد يدوي'}
                </div>
              </div>
            </button>
          </div>

          {/* Custom Currency Input */}
          {isCustomCurrency && (
            <div className="p-3.5 rounded-xl border border-dashed border-emerald-300 dark:border-emerald-800/70 bg-emerald-50/40 dark:bg-emerald-950/20 space-y-3">
              <div className="flex flex-col sm:flex-row gap-2.5 items-end">
                <div className="flex-1 w-full">
                  <Input
                    label="رمز أو اسم العملة المخصصة"
                    placeholder="مثال: ج.م أو د.ك أو QAR أو ليرة"
                    value={customCurrencyInput}
                    onChange={(e) => setCustomCurrencyInput(e.target.value)}
                    maxLength={20}
                    helperText="سيظهر هذا الرمز بجانب جميع المبالغ المالية في التطبيق."
                  />
                </div>
                <Button
                  onClick={handleSaveCustomCurrency}
                  disabled={!customCurrencyInput.trim() || savingCurrency}
                  variant="primary"
                  className="w-full sm:w-auto"
                >
                  {savingCurrency ? 'جاري الحفظ...' : 'تطبيق العملة'}
                </Button>
              </div>
            </div>
          )}

          <div className="flex items-center gap-2 pt-1 text-xs text-slate-500 dark:text-slate-400">
            <Coins className="w-4 h-4 text-emerald-500 shrink-0" />
            <span>
              العملة النشطة الحالية:{' '}
              <span className="font-bold text-emerald-600 dark:text-emerald-400">
                {activeConfig.name} ({activeConfig.symbol})
              </span>
            </span>
          </div>
        </div>
      </Card>

      {/* PWA App Installation Card */}
      <Card
        title="تثبيت التطبيق (PWA)"
        subtitle="يمكنك تثبيت تطبيق مصاريفي على هاتفك أو حاسوبك للعمل بدون إنترنت وسرعة فائقة"
      >
        <div className="pt-2 flex flex-col sm:flex-row sm:items-center justify-between gap-4">
          <div className="flex items-center gap-3">
            <div className="w-10 h-10 rounded-xl bg-emerald-50 dark:bg-emerald-950/40 text-emerald-600 dark:text-emerald-400 flex items-center justify-center">
              <Download className="w-5 h-5" />
            </div>
            <div>
              <div className="text-sm font-bold text-slate-800 dark:text-slate-200">
                {isInstalled ? 'التطبيق مثبت حالياً' : 'تطبيق ويب تقدمي (PWA)'}
              </div>
              <div className="text-xs text-slate-400">
                {isInstalled
                  ? 'يعمل التطبيق كبرنامج مستقل مع دعم كامل للعمل بلا اتصال'
                  : 'يعمل بكامل ميزاته مباشرة من الشاشة الرئيسية دون الحاجة للمتجر'}
              </div>
            </div>
          </div>

          {isInstalled ? (
            <div className="inline-flex items-center gap-1.5 px-3 py-1.5 rounded-xl bg-emerald-50 text-emerald-700 dark:bg-emerald-950/60 dark:text-emerald-300 text-xs font-bold">
              <CheckCircle2 className="w-4 h-4" />
              <span>مثبت على الجهاز</span>
            </div>
          ) : isInstallable ? (
            <Button
              variant="primary"
              size="sm"
              icon={<Download className="w-4 h-4" />}
              onClick={promptInstall}
            >
              تثبيت التطبيق على جهازك
            </Button>
          ) : (
            <span className="text-xs text-slate-400 font-medium">
              متاح للتثبيت من متصفحك عبر زر المشاركة أو القائمة
            </span>
          )}
        </div>
      </Card>

      {/* Data Export & Backup Card */}
      <Card
        title="تصدير البيانات والنسخ الاحتياطي"
        subtitle="تصدير بياناتك المالية وسجلات الديون إلى ملفات Excel/CSV أو نسخ احتياطي شامل JSON"
      >
        <div className="pt-2 space-y-5">
          {/* Section 1: Comprehensive JSON Backup */}
          <div className="p-4 rounded-2xl bg-emerald-50/50 dark:bg-emerald-950/20 border border-emerald-200 dark:border-emerald-800/50 space-y-3">
            <div className="flex items-center gap-2.5">
              <div className="w-8 h-8 rounded-xl bg-emerald-600 text-white flex items-center justify-center shrink-0">
                <Database className="w-4 h-4" />
              </div>
              <div>
                <h4 className="text-xs sm:text-sm font-bold text-slate-800 dark:text-slate-100">
                  النسخ الاحتياطي الشامل (JSON)
                </h4>
                <p className="text-[11px] text-slate-500 dark:text-slate-400">
                  حفظ أو استعادة كافة بيانات حسابك (المعاملات، التصنيفات، الديون، المتكررات، الجمعيات، والإعدادات) في ملف واحد.
                </p>
              </div>
            </div>

            <div className="flex flex-col sm:flex-row gap-2.5 pt-1">
              <Button
                variant="primary"
                size="md"
                icon={<Download className="w-4 h-4" />}
                onClick={handleDownloadJsonBackup}
                disabled={exportingJson || !user}
                className="flex-1 font-bold text-xs sm:text-sm"
              >
                {exportingJson ? 'جاري التحضير...' : 'تنزيل نسخة احتياطية شاملة (JSON)'}
              </Button>

              <Button
                variant="outline"
                size="md"
                icon={<Upload className="w-4 h-4 text-emerald-600 dark:text-emerald-400" />}
                onClick={handleTriggerFileInput}
                disabled={restoringJson || !user}
                className="flex-1 font-bold text-xs sm:text-sm"
              >
                استعادة نسخة احتياطية (JSON)
              </Button>

              <input
                ref={fileInputRef}
                type="file"
                accept=".json,application/json"
                onChange={handleFileSelected}
                className="hidden"
                aria-label="اختيار ملف النسخة الاحتياطية"
              />
            </div>
          </div>

          {/* Section 2: CSV Exports */}
          <div className="space-y-3">
            <h4 className="text-xs font-bold text-slate-700 dark:text-slate-300">
              تصدير الجداول (Excel / CSV مع دعم العربية):
            </h4>
            <div className="flex flex-col sm:flex-row gap-3">
              <Button
                variant="outline"
                size="md"
                icon={<FileSpreadsheet className="w-4 h-4 text-emerald-600 dark:text-emerald-400" />}
                onClick={handleExportTransactions}
                disabled={exportingTxs || !user}
                className="flex-1 font-bold"
              >
                {exportingTxs ? 'جاري التصدير...' : 'تصدير جميع المعاملات (CSV)'}
              </Button>

              <Button
                variant="outline"
                size="md"
                icon={<FileSpreadsheet className="w-4 h-4 text-emerald-600 dark:text-emerald-400" />}
                onClick={handleExportDebts}
                disabled={exportingDebts || !user}
                className="flex-1 font-bold"
              >
                {exportingDebts ? 'جاري التصدير...' : 'تصدير سجل الديون (CSV)'}
              </Button>
            </div>
          </div>
        </div>
      </Card>

      {/* Account Info */}
      <Card title="الحساب الشخصي" subtitle="معلومات تسجيل الدخول عبر Google">
        <div className="pt-2 flex flex-col sm:flex-row sm:items-center justify-between gap-4">
          <div className="flex items-center gap-3">
            {user?.photoURL ? (
              <img
                src={user.photoURL}
                alt={user.displayName || 'صورة المستخدم'}
                className="w-10 h-10 rounded-full object-cover border border-slate-200 dark:border-slate-700"
                referrerPolicy="no-referrer"
              />
            ) : (
              <div className="w-10 h-10 rounded-full bg-slate-200 dark:bg-slate-700 text-slate-700 dark:text-slate-200 flex items-center justify-center font-bold">
                {user?.displayName ? user.displayName.charAt(0) : 'م'}
              </div>
            )}
            <div>
              <div className="text-sm font-bold text-slate-800 dark:text-slate-200">
                {user?.displayName || 'مستخدم مصاريفي'}
              </div>
              <div className="text-xs text-slate-400" dir="ltr">
                {user?.email || 'بدون بريد إلكتروني'}
              </div>
            </div>
          </div>

          <Button
            variant="outline"
            size="sm"
            icon={<LogOut className="w-4 h-4 rtl:rotate-180" />}
            onClick={handleSignOut}
            disabled={loggingOut}
          >
            {loggingOut ? 'جاري الخروج...' : 'تسجيل الخروج'}
          </Button>
        </div>
      </Card>

      {/* Restore Confirmation Modal */}
      <Modal
        isOpen={Boolean(restoreValidation?.valid)}
        onClose={() => setRestoreValidation(null)}
        title="تأكيد استعادة النسخة الاحتياطية"
      >
        <div className="space-y-4">
          <p className="text-xs sm:text-sm text-slate-600 dark:text-slate-300 leading-relaxed">
            تم فحص ملف النسخة الاحتياطية بنجاح. فيما يلي ملخص البيانات التي سيتم استعادتها وحفظها في حسابك:
          </p>

          {restoreValidation?.summary && (
            <div className="p-3.5 rounded-2xl bg-slate-50 dark:bg-slate-900 border border-slate-200 dark:border-slate-800 space-y-2 text-xs">
              <div className="flex justify-between items-center py-1 border-b border-slate-200/60 dark:border-slate-800">
                <span className="text-slate-500">تاريخ إنشاء النسخة:</span>
                <span className="font-bold text-slate-800 dark:text-slate-200" dir="ltr">

/**
 * Checks for all due recurring transactions up to `today`,
 * creates idempotent transactions with deterministic IDs, and updates recurring items.
 * Returns the number of new transactions created.
 */
export async function processDueRecurring(
  uid: string,
  today: Date = new Date()
): Promise<number> {
  const endOfToday = new Date(today.getTime())
  endOfToday.setHours(23, 59, 59, 999)

  const colRef = collection(db, 'users', uid, 'recurring').withConverter(recurringConverter)
  const q = query(
    colRef,
    where('active', '==', true),
    where('nextDate', '<=', Timestamp.fromDate(endOfToday))
  )

  const snap = await getDocs(q)
  if (snap.empty) {
    return 0
  }

  let createdCount = 0
  let batchHasWrites = false
  const batch = writeBatch(db)

  for (const docSnap of snap.docs) {
    const rec = docSnap.data()
    const { dates, newNextDate, shouldDeactivate } = calculateDueOccurrences(rec, endOfToday)

    if (dates.length > 0) {
      for (const occDate of dates) {
        const txId = getDeterministicTxId(rec.id, occDate)
        const txRef = doc(db, 'users', uid, 'transactions', txId)
        const txSnap = await getDoc(txRef)

        if (!txSnap.exists()) {
          const newTxData: Record<string, unknown> = {
            type: rec.type,
            amountFils: rec.amountFils,
            categoryId: rec.categoryId,
            date: Timestamp.fromDate(occDate),
            note: typeof rec.note === 'string' ? rec.note.slice(0, 500) : '',
            recurringId: rec.id,
            createdAt: serverTimestamp(),
          }
          if (rec.title) {
            newTxData.title = rec.title.slice(0, 100)
          }
          batch.set(txRef, newTxData)
          createdCount++
        }
      }
    }

    const recRef = doc(db, 'users', uid, 'recurring', rec.id)
    const updatePayload: Record<string, unknown> = {
      nextDate: Timestamp.fromDate(newNextDate),
    }
    if (shouldDeactivate) {
      updatePayload.active = false
    }
    batch.update(recRef, updatePayload)
    batchHasWrites = true
  }

  if (batchHasWrites) {
    await batch.commit()
  }

  return createdCount
}

/**

 */
export async function syncCategoriesForUser(uid: string): Promise<void> {
  const categoriesRef = collection(db, 'users', uid, 'categories')
  const snap = await getDocs(categoriesRef)

  if (snap.empty) {
    await seedDefaultCategories(uid)
    return
  }

  const existingDocs = snap.docs.map((d) => ({ id: d.id, ...d.data() } as { id: string; name: string; type: TxType; order?: number }))
  const existingNames = new Set(existingDocs.map((c) => c.name))
  const batch = writeBatch(db)
  let changed = false

  // Handle renames
  for (const docItem of existingDocs) {
    if (docItem.name === 'تسوق وملابس') {
      const ref = doc(categoriesRef, docItem.id)
      batch.update(ref, { name: 'ملابس وإكسسوارات' })
      existingNames.delete('تسوق وملابس')
      existingNames.add('ملابس وإكسسوارات')
      changed = true
    } else if (docItem.name === 'عائلة وهدايا') {
      const ref = doc(categoriesRef, docItem.id)
      batch.update(ref, { name: 'الهدايا والمناسبات' })
      existingNames.delete('عائلة وهدايا')
      existingNames.add('الهدايا والمناسبات')
      changed = true
    }
  }

  // Handle missing new categories
  let maxOrder = existingDocs.reduce((acc, c) => Math.max(acc, c.order ?? 0), 0)

  for (const defaultCat of DEFAULT_CATEGORIES) {
    if (!existingNames.has(defaultCat.name)) {
      maxOrder++
      const newDocRef = doc(categoriesRef)
      batch.set(newDocRef, {
        name: defaultCat.name,
        type: defaultCat.type,
        icon: defaultCat.icon,
        color: defaultCat.color,
        archived: false,
        order: maxOrder,
      })
      changed = true
    }
  }

  if (changed) {
    await batch.commit()
  }
}

import { lazy } from 'react'
import { BrowserRouter, Routes, Route, Navigate } from 'react-router'
import { AppShell } from './components/layout/AppShell'
import { DashboardPage } from './pages/DashboardPage'
import { TransactionsPage } from './pages/TransactionsPage'
import { LoginPage } from './pages/LoginPage'
import { ProtectedRoute } from './components/ProtectedRoute'
import { AuthProvider } from './hooks/useAuth'
import { AppDataProvider } from './context/AppDataContext'
import { ToastProvider } from './hooks/useToast'
import { SelectedMonthProvider } from './hooks/useSelectedMonth'
import { TransactionModalProvider } from './hooks/useTransactionModal'
import { ToastContainer } from './components/ui/Toast'
import { useTheme } from './hooks/useTheme'
import { useRecurringProcessor } from './hooks/useRecurringProcessor'

// Lazy loaded secondary/heavy pages to prevent blocking the initial bundle
const DebtsPage = lazy(() => import('./pages/DebtsPage').then((m) => ({ default: m.DebtsPage })))
const RecurringPage = lazy(() => import('./pages/RecurringPage').then((m) => ({ default: m.RecurringPage })))
const ReportsPage = lazy(() => import('./pages/ReportsPage').then((m) => ({ default: m.ReportsPage })))
const CategoriesPage = lazy(() => import('./pages/CategoriesPage').then((m) => ({ default: m.CategoriesPage })))
const SettingsPage = lazy(() => import('./pages/SettingsPage').then((m) => ({ default: m.SettingsPage })))
const ToolsPage = lazy(() => import('./pages/ToolsPage').then((m) => ({ default: m.ToolsPage })))

function AppRoutes() {
  // Initialize and sync theme with user settings and localStorage
  useTheme()

  // Process any due recurring transactions when user is logged in
  useRecurringProcessor()

  return (
    <Routes>
      {/* Auth Route without shell (redirects to / if already logged in) */}
      <Route path="/login" element={<LoginPage />} />

      {/* Protected Application Routes wrapped in AppShell */}
      <Route
        element={
          <ProtectedRoute>
            <AppShell />
          </ProtectedRoute>
        }
      >
        <Route index element={<DashboardPage />} />
        <Route path="transactions" element={<TransactionsPage />} />
        <Route path="debts" element={<DebtsPage />} />
        <Route path="tools" element={<ToolsPage />} />
        <Route path="recurring" element={<RecurringPage />} />
        <Route path="reports" element={<ReportsPage />} />
        <Route path="categories" element={<CategoriesPage />} />
        <Route path="settings" element={<SettingsPage />} />
      </Route>

      {/* Fallback */}
      <Route path="*" element={<Navigate to="/" replace />} />
    </Routes>
  )
}

export function App() {
  return (
    <BrowserRouter>
      <AuthProvider>
        <AppDataProvider>
          <ToastProvider>
            <SelectedMonthProvider>
              <TransactionModalProvider>
                <AppRoutes />
                <ToastContainer />
              </TransactionModalProvider>
            </SelectedMonthProvider>
          </ToastProvider>
        </AppDataProvider>
      </AuthProvider>
    </BrowserRouter>
  )
}

export default App
